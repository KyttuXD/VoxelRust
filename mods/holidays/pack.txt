# Holidays - a mod that comes with the game, in its own folder.
#
# Decorations and treats for the year's feasts: winter holidays, the
# harvest festival, the spooky season, the new year and spring. Everything
# is made at a crafting table or in an oven; nothing is handed out.
# Switch the mod off in Settings > MODS, or delete the folder.

# ---- Big props ----
# One block each, but several blocks wide and tall: `prop` names the model
# and how tall to draw it, `fills` how much room it takes (blocks wide,
# blocks tall). The room must be free to put one down, the whole of it is
# solid, and breaking any part takes the lot. Models: tools/gen_props.py.
model grand_tree file models/grand_tree.bbmodel
model giant_snowman file models/giant_snowman.bbmodel
model great_pumpkin file models/great_pumpkin.bbmodel
model maypole file models/maypole.bbmodel

block grand_tree
  name Grand Festive Tree
  tex fir_needles
  light 12
  hardness 2
  box 0 0 0 16 16 16
  prop grand_tree 7
  fills 3 7
  category Holidays
block giant_snowman
  name Giant Snowman
  tex snow
  hardness 1
  rotate
  box 0 0 0 16 16 16
  prop giant_snowman 3.5
  fills 1 3
  category Holidays
block great_pumpkin
  name Great Pumpkin
  tex jack_face
  light 13
  hardness 2
  rotate
  box 0 0 0 16 16 16
  prop great_pumpkin 3
  fills 3 3
  category Holidays
block maypole
